function extractImageId(url) {
  if (!url) return null;
  const match = url.match(/(?:id=|\/d\/)([a-zA-Z0-9_-]+)/);
  return match ? match[1] : null;
}

function parseMetadataDoc(docId) {
  let meta = {};
  let desc = "";
  try {
    const text = DocumentApp.openById(docId).getBody().getText();
    let lines = text.split('\n');
    let currentKey = null;

    for (let i = 0; i < lines.length; i++) {
      let line = lines[i];
      let match = line.match(/^([A-Za-z0-9 ]+):\s*(.*)$/);
      
      if (match) {
        let key = match[1].trim().toLowerCase();
        let val = match[2].trim();
        
        if (key === 'description') {
           let descLines = [];
           if (val) descLines.push(val);
           let j = i + 1;
           // Keep reading until we hit another key like "Instagram Handles:"
           while (j < lines.length) {
              if (lines[j].match(/^([A-Za-z0-9 ]+):\s*(.*)$/)) break;
              descLines.push(lines[j]);
              j++;
           }
           desc = descLines.join('<br>').trim();
           i = j - 1; // Fast-forward the main loop
        } else if (key === 'instagram handles' || key === 'instagram') {
           let instaLines = [];
           if (val) instaLines.push(val);
           let j = i + 1;
           while (j < lines.length) {
              if (lines[j].match(/^([A-Za-z0-9 ]+):\s*(.*)$/)) break;
              instaLines.push(lines[j]);
              j++;
           }
           meta['instagram handles'] = instaLines.join('<br>').trim();
           i = j - 1;
        } else {
           if (key === 'note') key = 'author';
           if (key === 'event') key = 'course';
           if (key === 'date') key = 'dateoverride';
           currentKey = key;
           meta[currentKey] = val;
        }
      } else if (currentKey && line.trim() !== "") {
        meta[currentKey] += " " + line.trim();
      }
    }
    
    if (meta.tags && typeof meta.tags === 'string') {
       meta.tags = meta.tags.split(',').map(t => t.trim()).filter(t => t);
    }
  } catch(e) {}
  
  return { meta: meta, desc: desc };
}

        const name = file.getName().toLowerCase();
        const url = "https://drive.google.com/uc?export=view&id=" + file.getId();
        
        if (name.includes('title')) {
          projectData.titleImage = url;
        } else if (mime.includes('image')) {
          let cleanName = file.getName().split('.').slice(0, -1).join('.') || file.getName();
          projectData.images.push({
            src: url,
            title: cleanName
          });

        else if (mime === MimeType.GOOGLE_DOCS) {
          let parsed = parseMetadataDoc(file.getId());
          for (let k in parsed.meta) projectData.metadata[k] = parsed.meta[k];
          if (parsed.desc) projectData.metadata.description = parsed.desc;
        }
      }
      if (projectData.titleImage) projects.push(projectData);
    }
  } catch (err) {}

function buildHistoryData() {
  try {
    const folder = DriveApp.getFolderById(FOLDERS.HISTORY);
    const files = folder.getFilesByType(MimeType.GOOGLE_DOCS);
    let historyDocs = [];
    while (files.hasNext()) {
      const file = files.next();
      const doc = DocumentApp.openById(file.getId());
      let rawName = file.getName();
      let cleanTitle = rawName.replace(/^\d+[_ -]/, '').trim(); 
      let text = doc.getBody().getText().replace(/\n/g, '<br>');
      if (text) historyDocs.push({ rawName: rawName, title: cleanTitle, text: text });
    }
    historyDocs.sort((a, b) => a.rawName.localeCompare(b.rawName));
    return historyDocs.map(d => ({ title: d.title, text: d.text }));
  } catch (err) { return []; }
}

  try {
    // This will overwrite the entire file with the latest JSON data
    DriveApp.getFileById(DRIVE_JSON_FILE_ID).setContent(jsonStr);

    console.error("Failed to update Drive file: " + e);
  }
  // -------------------------------------------------
  
  // -- GITHUB PUSH LOGIC --
  const GITHUB_TOKEN = PropertiesService.getScriptProperties().getProperty('GITHUB_PAT');
  const REPO = 'florianroedel-hash/apollo';
  const FILE_PATH = 'data.json';
  
  if (!GITHUB_TOKEN) {
    safeAlert("❌ Missing GitHub Token", "Please add a GITHUB_PAT token in your Script Properties.");
    return;
  }
  
  try {
    const apiUrl = `https://api.github.com/repos/${REPO}/contents/${FILE_PATH}`;
    const headers = {
      "Authorization": "Bearer " + GITHUB_TOKEN,
      "Accept": "application/vnd.github.v3+json",
      "X-GitHub-Api-Version": "2022-11-28"
    };
    
    // 1. Get current file SHA to update it (required by GitHub API)
    let sha = null;
    const getRes = UrlFetchApp.fetch(apiUrl, { method: "get", headers: headers, muteHttpExceptions: true });
    if (getRes.getResponseCode() === 200) {
      const currentFile = JSON.parse(getRes.getContentText());
      sha = currentFile.sha;
    }
    
    // 2. Put new file
    const contentEncoded = Utilities.base64Encode(Utilities.newBlob(jsonStr).getBytes());
    const putPayload = {
      message: "Automated publish from Google Sheets",
      content: contentEncoded,
      branch: "main"
    };
    if (sha) putPayload.sha = sha;
    
    const putRes = UrlFetchApp.fetch(apiUrl, {
      method: "put",
      headers: headers,
      contentType: "application/json",
      payload: JSON.stringify(putPayload),
      muteHttpExceptions: true
    });
    
    if (putRes.getResponseCode() === 200 || putRes.getResponseCode() === 201) {
       // Save to Cache Sheet so Inventory can verify what is currently live
       const CHUNK_SIZE = 40000; 
       let chunks = [];
       for (let i = 0; i < jsonStr.length; i += CHUNK_SIZE) chunks.push([jsonStr.substring(i, i + CHUNK_SIZE)]);
       let cacheSheet = ss.getSheetByName("System_Cache");
       if (!cacheSheet) { cacheSheet = ss.insertSheet("System_Cache"); cacheSheet.hideSheet(); }
       cacheSheet.clear();
       cacheSheet.getRange(1, 1, chunks.length, 1).setValues(chunks);
       
       scanLiveInventory();
       safeAlert("✅ Publish Successful!", "Your JSON data was pushed to GitHub and updated on Google Drive! The automated MP3 Delivery Guy is now downloading your audio files.");
    } else {
       safeAlert("❌ Publish Failed", "GitHub Error: " + putRes.getContentText());
    }
  } catch(e) {
    safeAlert("❌ Publish Error", e.toString());
  }
}

function updateCalendarVisuals(ss) {
  const calSheet = ss.getSheetByName("Calendar");
  if (!calSheet) return;
  const data = calSheet.getDataRange().getValues();
  
  for (let i = data.length; i >= 1; i--) {
    let dateStr = String(calSheet.getRange(i, 1).getValue());
    if (dateStr.includes('-------')) calSheet.deleteRow(i);
  }
  
  const newData = calSheet.getDataRange().getValues();
  const now = new Date();
  now.setHours(0,0,0,0);
  
  let insertIndex = -1;
  let lastWasPast = null;
  
  for (let i = 1; i < newData.length; i++) {
    let raw = newData[i][0];
    if (!raw) continue;
    if (String(raw).toLowerCase().includes('marquee')) continue;
    
    let parsed = null;
    if (raw instanceof Date) parsed = raw;
    else {
       let m = String(raw).match(/(\d{1,2})[\.\/ -](\d{1,2})[\.\/ -](\d{2,4})/);
       if (m) {
         let yr = parseInt(m[3]);
         if (yr < 100) yr += 2000;
         parsed = new Date(yr, parseInt(m[2])-1, m[1]);
       } else parsed = new Date(raw);
    }
    
    if (parsed && !isNaN(parsed.getTime())) {
      let isPast = parsed < now;
      if (lastWasPast !== null && lastWasPast !== isPast) {
         insertIndex = i + 1;
         break;
      }
      lastWasPast = isPast;
    }
  }
  
  if (insertIndex > -1) {
    calSheet.insertRowBefore(insertIndex);
    calSheet.getRange(insertIndex, 1).clearDataValidations();
    calSheet.getRange(insertIndex, 1, 1, 3).setValues([["-------", "-------", "-------"]]);
    calSheet.getRange(insertIndex, 1, 1, 3).setBackground("#fff2cc");
  }

function extractImageId(url) {
  if (!url) return null;
  const match = url.match(/(?:id=|\/d\/)([a-zA-Z0-9_-]+)/);
  return match ? match[1] : null;
}

// --- NEW SMART PARSER HELPER ---
function parseMetadataDoc(docId) {
  let meta = {};
  let desc = "";
  try {
    const text = DocumentApp.openById(docId).getBody().getText();
    let lines = text.split('\n');
    let currentKey = null;

    for (let i = 0; i < lines.length; i++) {
      let line = lines[i];
      let match = line.match(/^([A-Za-z0-9 ]+):\s*(.*)$/);
      
      if (match) {
        let key = match[1].trim().toLowerCase();
        let val = match[2].trim();
        
        if (key === 'description') {
           let descLines = [];
           if (val) descLines.push(val);
           let j = i + 1;
           // Keep reading until we hit another key like "Instagram Handles:"
           while (j < lines.length) {
              if (lines[j].match(/^([A-Za-z0-9 ]+):\s*(.*)$/)) break;
              descLines.push(lines[j]);
              j++;
           }
           desc = descLines.join('<br>').trim();
           i = j - 1; // Fast-forward the main loop
        } else if (key === 'instagram handles' || key === 'instagram') {
           let instaLines = [];
           if (val) instaLines.push(val);
           let j = i + 1;
           while (j < lines.length) {
              if (lines[j].match(/^([A-Za-z0-9 ]+):\s*(.*)$/)) break;
              instaLines.push(lines[j]);
              j++;
           }
           meta['instagram handles'] = instaLines.join('<br>').trim();
           i = j - 1;
        } else {
           if (key === 'note') key = 'author';
           if (key === 'event') key = 'course';
           if (key === 'date') key = 'dateoverride';
           currentKey = key;
           meta[currentKey] = val;
        }
      } else if (currentKey && line.trim() !== "") {
        meta[currentKey] += " " + line.trim();
      }
    }
    
    if (meta.tags && typeof meta.tags === 'string') {
       meta.tags = meta.tags.split(',').map(t => t.trim()).filter(t => t);
    }
  } catch(e) {}
  
  return { meta: meta, desc: desc };
}

// --- CALENDAR (FROM SHEET) ---
function buildCalendarData(ss) {
  const sheet = ss.getSheetByName("Calendar");
  if (!sheet) return [];
  const rawData = sheet.getDataRange().getValues();
  const displayData = sheet.getDataRange().getDisplayValues();
  
  let events = [];
  let pastUpcomingLists = { past: [], upcoming: [] };
  const now = new Date();
  now.setHours(0,0,0,0);
  let statsUpdates = [];
  
  for (let i = 1; i < displayData.length; i++) {
    let dateObj = rawData[i][0]; 
    let dateStr = displayData[i][0];
    let details = displayData[i][1];
    let link = displayData[i][2]; 
    
    if (dateStr && dateStr.includes('-------')) {
      statsUpdates.push([""]);
      continue;
    }
    
    let isPast = false;
    let parsedDate = null;
    let statText = "";
    
    if (dateObj instanceof Date) parsedDate = dateObj;
    else {
       let matchEu = String(dateStr).match(/(\d{1,2})[\.\/ -](\d{1,2})[\.\/ -](\d{2,4})/);
       if (matchEu) {
         let yr = parseInt(matchEu[3]);
         if (yr < 100) yr += 2000;
         parsedDate = new Date(yr, parseInt(matchEu[2])-1, matchEu[1]);
       } else parsedDate = new Date(dateStr);
    }

    if (parsedDate && !isNaN(parsedDate.getTime()) && parsedDate < now) isPast = true;
    
    let imgId = null;
    let spreadData = null;
    
    if (link) {
      let extractedId = null;
      if (link.includes('drive.google.com')) extractedId = extractImageId(link);
      else extractedId = link.trim();
      
      if (extractedId) {
        try {
          const folder = DriveApp.getFolderById(extractedId);
          const files = folder.getFiles();
          let fileList = [];
          
          let evMeta = { name: dateStr, dateOverride: dateStr, year: "2026", author: "Apollo", course: "", track: "", description: details, tags: ["Event"] };
          
          while (files.hasNext()) {
            let f = files.next();
            if (f.getMimeType().includes('image') || f.getMimeType() === MimeType.PDF) {
               try { f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch(e){}
               fileList.push({name: f.getName(), url: "https://drive.google.com/uc?id=" + f.getId()});
            } else if (f.getMimeType() === MimeType.GOOGLE_DOCS) {
               let parsed = parseMetadataDoc(f.getId());
               for (let k in parsed.meta) {
                 if (k === 'name' || k === 'title') evMeta.dateOverride = parsed.meta[k]; 
                 else evMeta[k] = parsed.meta[k];
               }
               if (parsed.desc) evMeta.description = parsed.desc;
            }
          }
          fileList.sort((a,b) => a.name.localeCompare(b.name));
          statText = fileList.length + " files found";
          
          if (fileList.length > 0) {
            imgId = extractImageId(fileList[0].url); 
            if (isPast) {
              evMeta.tags = ["Event"]; 
              spreadData = {
                id: "calevt_" + i,
                metadata: evMeta,
                titleImage: fileList[0].url,
                images: fileList.length > 1 ? fileList.slice(1).map(f => {
                   let cleanName = f.name.split('.').slice(0, -1).join('.') || f.name;
                   return { src: f.url, title: cleanName };
                }) : []
              };
            }
          } else {
            imgId = extractedId;
            try { DriveApp.getFileById(imgId).setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch(e){}
            statText = "1 direct link";
          }
        } catch(e) { 
          imgId = extractedId; 
          statText = "1 direct link";
          try { DriveApp.getFileById(imgId).setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch(e){}
        }
      } else statText = "Invalid link";
    } else {
      if (dateStr && !dateStr.toLowerCase().includes('marquee')) statText = "No images";
    }
    
    let eventObj = { isPast: isPast, timestamp: (parsedDate && !isNaN(parsedDate.getTime())) ? parsedDate.getTime() : null, text: dateStr + "<br>" + details, imgId: imgId, spreadData: spreadData };
    if (isPast) pastUpcomingLists.past.push(eventObj);
    else if (dateStr && !dateStr.toLowerCase().includes('marquee')) pastUpcomingLists.upcoming.push(eventObj);
    statsUpdates.push([statText]);
  }
  
  if (statsUpdates.length > 0) sheet.getRange(2, 4, statsUpdates.length, 1).setValues(statsUpdates);
  
  pastUpcomingLists.upcoming.sort((a, b) => b.timestamp - a.timestamp);
  pastUpcomingLists.past.sort((a, b) => b.timestamp - a.timestamp);
  events = pastUpcomingLists.upcoming.concat(pastUpcomingLists.past);
  
  let marquees = [];
  const marqSheet = ss.getSheetByName('Marquee');
  if (marqSheet) {
    const mRaw = marqSheet.getDataRange().getValues();
    const mDisplay = marqSheet.getDataRange().getDisplayValues();
    for (let i = 1; i < mRaw.length; i++) {
      let msg = String(mDisplay[i][0]).trim();
      if (!msg) continue;
      let marqueeObj = { message: msg };

    if (isPast) pastUpcomingLists.past.push(eventObj);
    else if (dateStr && !dateStr.toLowerCase().includes('marquee')) pastUpcomingLists.upcoming.push(eventObj);
    statsUpdates.push([statText]);
  }
  
  if (statsUpdates.length > 0) sheet.getRange(2, 4, statsUpdates.length, 1).setValues(statsUpdates);
  
  pastUpcomingLists.upcoming.sort((a, b) => b.timestamp - a.timestamp);
  pastUpcomingLists.past.sort((a, b) => b.timestamp - a.timestamp);
  events = pastUpcomingLists.upcoming.concat(pastUpcomingLists.past);
  
  let marquees = [];
  const marqSheet = ss.getSheetByName('Marquee');
  if (marqSheet) {
    const mRaw = marqSheet.getDataRange().getValues();
    const mDisplay = marqSheet.getDataRange().getDisplayValues();
    for (let i = 1; i < mRaw.length; i++) {
      let msg = String(mDisplay[i][0]).trim();
      if (!msg) continue;
      let marqueeObj = { message: msg };
      if (mRaw[i][1]) marqueeObj.startTime = (mRaw[i][1] instanceof Date) ? mRaw[i][1].toISOString() : String(mDisplay[i][1]).trim();
      if (mRaw[i][2]) marqueeObj.endTime = (mRaw[i][2] instanceof Date) ? mRaw[i][2].toISOString() : String(mDisplay[i][2]).trim();
      if (mRaw[i][3] instanceof Date) marqueeObj.startDate = mRaw[i][3].getFullYear() + "-" + String(mRaw[i][3].getMonth()+1).padStart(2,'0') + "-" + String(mRaw[i][3].getDate()).padStart(2,'0');
      else if (mDisplay[i][3]) marqueeObj.startDate = String(mDisplay[i][3]).trim();
      if (mRaw[i][4] instanceof Date) marqueeObj.endDate = mRaw[i][4].getFullYear() + "-" + String(mRaw[i][4].getMonth()+1).padStart(2,'0') + "-" + String(mRaw[i][4].getDate()).padStart(2,'0');
      else if (mDisplay[i][4]) marqueeObj.endDate = String(mDisplay[i][4]).trim();
      if (mDisplay[i][5]) marqueeObj.dayOfWeek = String(mDisplay[i][5]).trim();
      marquees.push(marqueeObj);
    }
  }
  return [{ events: events, marquees: marquees }];
}

                 // Push the audio to Github natively!
                 audioMap[data[i][0]] = uploadAudioToGitHub(String(data[i][1]).trim());
              }
            }
          }
          break; // Stop looking after we find the index
        }
      }
      
      const files = folder.getFiles();
      while (files.hasNext()) {

        const mime = file.getMimeType();

        let audioUrl = audioMap[name] || null;
        
        if (name.toLowerCase().includes('title')) {
          projectData.titleImage = url;
          if (audioUrl) projectData.titleAudio = audioUrl; // Attaches audio to the title image
        } else if (mime.includes('image')) {
          let cleanName = name.split('.').slice(0, -1).join('.') || name;
          let imgObj = { src: url, title: cleanName };
          if (audioUrl) imgObj.audioUrl = audioUrl; // Attaches audio to specific page/image
          projectData.images.push(imgObj);
        }
        else if (mime === MimeType.GOOGLE_DOCS) {
          let parsed = parseMetadataDoc(file.getId());
          for (let k in parsed.meta) projectData.metadata[k] = parsed.meta[k];
          if (parsed.desc) projectData.metadata.description = parsed.desc;
        }
      }
      if (projectData.titleImage) projects.push(projectData);
    }

function buildMagazineData() {
  try {
    const root = DriveApp.getFolderById(FOLDERS.MAGAZINE);
    let issuesPayload = [];
    let issueFolders = root.getFolders();
    let issueList = [];
    while(issueFolders.hasNext()) issueList.push(issueFolders.next());
    issueList.sort((a,b) => a.getName().localeCompare(b.getName()));
    const now = new Date(); now.setHours(0,0,0,0);
    
    for (let x = 0; x < issueList.length; x++) {
       let issueFolder = issueList[x], currentIssueTitle = issueFolder.getName();
       let files = issueFolder.getFilesByType(MimeType.GOOGLE_SHEETS);
       if (!files.hasNext()) continue;
       const sheetFile = files.next();
       const data = SpreadsheetApp.openById(sheetFile.getId()).getSheets()[0].getDataRange().getValues();
       
       let fileUrls = {};
       let issueFiles = issueFolder.getFiles();
       while(issueFiles.hasNext()) {
         let img = issueFiles.next();
         if(img.getMimeType().includes('image')) fileUrls[img.getName()] = "https://drive.google.com/uc?export=view&id=" + img.getId();
       }
       
       let subfolders = issueFolder.getFolders();
       while(subfolders.hasNext()) {
         let sub = subfolders.next(), imgs = sub.getFiles();
         while(imgs.hasNext()) {
           let img = imgs.next();
           if(img.getMimeType().includes('image')) fileUrls[img.getName()] = "https://drive.google.com/uc?export=view&id=" + img.getId();
         }
       }
       
       let currentImageFiles = [], currentChapters = [], currentPages = [], currentChapterName = "", globalPageIndex = 0;
       let comingSoonUrl = null, hasLockedPages = false;
       
       for (let fName in fileUrls) {
          if (fName.toLowerCase().replace(/[^a-z0-9]/g, '').includes('comingsoon')) { comingSoonUrl = fileUrls[fName]; break; }
       }
       
       for (let i = 1; i < data.length; i++) {
          let type = data[i][0], name = data[i][1], audioUrlRaw = data[i][2] ? String(data[i][2]).trim() : null, dateRaw = data[i][3];
          
          if (type === "📂 CHAPTER") currentChapterName = name; 
          else if (type === "📄 Page" && name) {
            let isLocked = false;
            if (dateRaw) {
              let parsedDate = new Date(dateRaw);
              if (!isNaN(parsedDate.getTime()) && parsedDate > now) isLocked = true;
            }
            if (isLocked) {
              hasLockedPages = true;
              if (!currentChapters.find(c => c.name === currentChapterName)) currentChapters.push({ name: currentChapterName, page: "LOCKED" });
              continue; 
            }
            let fileUrl = fileUrls[name];
            if (fileUrl) {
              currentImageFiles.push(fileUrl);
              let pageMeta = {}; 
              // Upload magazine audio to Github natively
              if (audioUrlRaw) pageMeta.audioUrl = uploadAudioToGitHub(audioUrlRaw);
              currentPages.push(pageMeta);
              if (!currentChapters.find(c => c.name === currentChapterName)) currentChapters.push({ name: currentChapterName, page: globalPageIndex });
              globalPageIndex++;
            }
          }
       }
       
       let finalComingSoonIndex = currentImageFiles.length; 
       if (hasLockedPages && comingSoonUrl) {
          currentImageFiles.push(comingSoonUrl); currentPages.push({}); 
          currentChapters.forEach(c => { if (c.page === "LOCKED") c.page = finalComingSoonIndex; });
       } else currentChapters = currentChapters.filter(c => c.page !== "LOCKED");
       
       if (currentImageFiles.length > 0) issuesPayload.push({ title: currentIssueTitle, images: currentImageFiles, chapters: currentChapters, pages: currentPages });
    }
    return issuesPayload;
  } catch (err) { return []; }
}

// --- UPDATED HISTORY (USING INDEX SHEET & SUBFOLDERS) ---
function buildHistoryData() {
  try {
    const folder = DriveApp.getFolderById(FOLDERS.HISTORY);
    const historyData = [];
    
    let indexSheetFile = null;
    const sheets = folder.getFilesByType(MimeType.GOOGLE_SHEETS);
    while (sheets.hasNext()) {
      let sheet = sheets.next();
      if (sheet.getName().toLowerCase() === "index") {
        indexSheetFile = sheet;
        break;
      }
    }
    
    if (!indexSheetFile) return historyData; 
    
    const ss = SpreadsheetApp.openById(indexSheetFile.getId());
    const sheet = ss.getSheets()[0]; 
    const lastRow = sheet.getLastRow();
    
    if (lastRow <= 1) return historyData; 
    
    const dataRange = sheet.getRange(2, 1, lastRow - 1, 4);
    const dataValues = dataRange.getValues();
    
    const docsInFolder = {};
    
    function scanFolderForDocs(currentFolder) {
      let docsFiles = currentFolder.getFilesByType(MimeType.GOOGLE_DOCS);
      while (docsFiles.hasNext()) {
        let docFile = docsFiles.next();
        docsInFolder[docFile.getName()] = docFile;
      }
      let subfolders = currentFolder.getFolders();
      while (subfolders.hasNext()) {
        scanFolderForDocs(subfolders.next());
      }
    }
    
    scanFolderForDocs(folder); 
    
    function makeDirectLink(url) {
      if (!url || typeof url !== "string") return "";
      let match = url.match(/[-\w]{25,}/); 
      if (match && url.includes("drive.google.com")) {
        return "https://drive.google.com/uc?export=view&id=" + match[0];
      }
      return url; 
    }
    
    for (let i = 0; i < dataValues.length; i++) {
      const row = dataValues[i];
      const dateVal = row[0];
      const nameVal = String(row[1]).trim();
      
      // Push history audio straight to GitHub natively!
      const audioUrlVal = uploadAudioToGitHub(row[2]);
      const imageUrlVal = makeDirectLink(row[3]);
      
      if (!nameVal) continue;
      
      const matchedDocFile = docsInFolder[nameVal];
      if (!matchedDocFile) {
        console.warn("History Doc missing for: " + nameVal);
        continue;
      }
      
      const doc = DocumentApp.openById(matchedDocFile.getId());
      const textContent = doc.getBody().getText().replace(/\n/g, '<br>');
      
      historyData.push({
        date: dateVal instanceof Date ? dateVal.toISOString() : dateVal,
        name: nameVal,
        audioUrl: audioUrlVal,
        imageUrl: imageUrlVal,
        text: textContent
      });
    }
    
    return historyData;
  } catch (err) { 
    return []; 
  }
}

function buildAudioData() {

        const cleanTrackName = trackFolder.getName().replace(/^\d+[_ -]/, '').trim();
        let trackObj = { category: cleanCategoryName, title: cleanTrackName, file: "", url: "", year: "2026", type: "Audio", format: "MP3", desc: "No description provided." };
        const files = trackFolder.getFiles();
        while (files.hasNext()) {
          const file = files.next();
          const mime = file.getMimeType();
          if (mime.includes('audio') || mime.includes('video')) { 
            // ----- UPLOAD TO GITHUB DIRECTLY -----
            let uploadedPath = uploadAudioToGitHub(file.getId());
            trackObj.file = uploadedPath; 
            trackObj.url = uploadedPath;
            // -------------------------------------
          } 
          else if (mime === MimeType.GOOGLE_DOCS) {
            let parsed = parseMetadataDoc(file.getId());
            if (parsed.meta.name || parsed.meta.title) trackObj.title = parsed.meta.name || parsed.meta.title;
            if (parsed.desc) trackObj.desc = parsed.desc;
          }
        }
        if (trackObj.file) audioData.push(trackObj);
      }
    }
    return audioData;
  } catch (err) { return []; }
}
